366	import * as THREE from 'three';
367	
368	export class Planet {
369	  constructor(radius = 15) {
370	    this.radius = radius;
371	    this.mesh = this.createPlanet();
372	  }
373	
374	  createPlanet() {
375	    const geometry = new THREE.SphereGeometry(this.radius, 64, 64);
376	    
377	    // Toon material
378	    const material = new THREE.MeshToonMaterial({
379	      color: 0x88dd88,
380	      gradientMap: this.createGradientMap()
381	    });
382	
383	    const mesh = new THREE.Mesh(geometry, material);
384	    mesh.receiveShadow = true;
385	    return mesh;
386	  }
387	
388	  createGradientMap() {
389	    const colors = new Uint8Array([0, 128, 255]);
390	    const gradientMap = new THREE.DataTexture(colors, 3, 1, THREE.RedFormat);
391	    gradientMap.minFilter = THREE.NearestFilter;
392	    gradientMap.magFilter = THREE.NearestFilter;
393	    gradientMap.needsUpdate = true;
394	    return gradientMap;
395	  }
396	
397	  getCenter() {
398	    return this.mesh.position.clone();
399	  }
400	
401	  getSurfaceNormal(position) {
402	    return position.clone()
403	      .sub(this.getCenter())
404	      .normalize();
405	  }
406	}
407	Player Class
408	javascript// game/Player.js
409	import * as THREE from 'three';
410	
411	export class Player {
412	  constructor(planet) {
413	    this.planet = planet;
414	    this.mesh = this.createCharacter();
415	    this.speed = 3;
416	    this.velocity = new THREE.Vector3();
417	    
418	    // Position on top of planet
419	    this.mesh.position.set(0, planet.radius + 1, 0);
420	  }
421	
422	  createCharacter() {
423	    // Simple capsule character for MVP
424	    const group = new THREE.Group();
425	    
426	    // Body
427	    const bodyGeometry = new THREE.CapsuleGeometry(0.3, 1, 8, 16);
428	    const bodyMaterial = new THREE.MeshToonMaterial({ color: 0xff6b6b });
429	    const body = new THREE.Mesh(bodyGeometry, bodyMaterial);
430	    body.position.y = 0.5;
431	    group.add(body);
432	    
433	    // Head
434	    const headGeometry = new THREE.SphereGeometry(0.4, 16, 16);
435	    const headMaterial = new THREE.MeshToonMaterial({ color: 0xffddaa });
436	    const head = new THREE.Mesh(headGeometry, headMaterial);
437	    head.position.y = 1.5;
438	    group.add(head);
439	    
440	    group.castShadow = true;
441	    return group;
442	  }
443	
444	  update(input, deltaTime) {
445	    // Movement direction in local space
446	    const moveDirection = new THREE.Vector3();
447	    if (input.forward) moveDirection.z -= 1;
448	    if (input.backward) moveDirection.z += 1;
449	    if (input.left) moveDirection.x -= 1;
450	    if (input.right) moveDirection.x += 1;
451	    
452	    if (moveDirection.length() > 0) {
453	      moveDirection.normalize();
454	      
455	      // Apply movement in character's local space
456	      const worldDirection = moveDirection.applyQuaternion(this.mesh.quaternion);
457	      this.mesh.position.add(worldDirection.multiplyScalar(this.speed * deltaTime));
458	    }
459	    
460	    // Stick to planet surface
461	    this.stickToPlanet();
462	  }
463	
464	  stickToPlanet() {
465	    const planetCenter = this.planet.getCenter();
466	    const surfaceNormal = this.mesh.position.clone()
467	      .sub(planetCenter)
468	      .normalize();
469	    
470	    // Set position at correct radius
471	    this.mesh.position.copy(
472	      planetCenter.add(surfaceNormal.multiplyScalar(this.planet.radius + 1))
473	    );
474	    
475	    // Align to surface
476	    const targetQuaternion = new THREE.Quaternion().setFromUnitVectors(
477	      new THREE.Vector3(0, 1, 0),
478	      surfaceNormal
479	    );
480	    this.mesh.quaternion.slerp(targetQuaternion, 0.2);
481	  }
482	}

485	export class InputManager {
486	  constructor() {
487	    this.keys = {
488	      forward: false,
489	      backward: false,
490	      left: false,
491	      right: false,
492	      action: false
493	    };
494	    
495	    this.setupListeners();
496	  }
497	
498	  setupListeners() {
499	    window.addEventListener('keydown', (e) => this.handleKeyDown(e));
500	    window.addEventListener('keyup', (e) => this.handleKeyUp(e));
501	  }
502	
503	  handleKeyDown(e) {
504	    switch(e.key.toLowerCase()) {
505	      case 'w': case 'arrowup': this.keys.forward = true; break;
506	      case 's': case 'arrowdown': this.keys.backward = true; break;
507	      case 'a': case 'arrowleft': this.keys.left = true; break;
508	      case 'd': case 'arrowright': this.keys.right = true; break;
509	      case 'e': case ' ': this.keys.action = true; break;
510	    }
511	  }
512	
513	  handleKeyUp(e) {
514	    switch(e.key.toLowerCase()) {
515	      case 'w': case 'arrowup': this.keys.forward = false; break;
516	      case 's': case 'arrowdown': this.keys.backward = false; break;
517	      case 'a': case 'arrowleft': this.keys.left = false; break;
518	      case 'd': case 'arrowright': this.keys.right = false; break;
519	      case 'e': case ' ': this.keys.action = false; break;
520	    }
521	  }
522	
523	  getInput() {
524	    return { ...this.keys };
525	  }
526	}
